function resizeScene() { const scene = document.querySelector("a-scene"); if (!scene) { return; } if (scene.resize) { scene.resize(); } if ( scene.renderer && scene.camera ) { const width = window.innerWidth; const height = window.innerHeight; scene.renderer.setSize( width, height, false ); scene.camera.aspect = width / height; scene.camera.updateProjectionMatrix(); } }
